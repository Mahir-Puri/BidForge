cmake_minimum_required(VERSION 3.20)
project(BidForge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ASAN "Address and undefined behavior sanitizers" OFF)
option(TSAN "Thread sanitizer (separate build)" OFF)
option(UBSAN "Undefined behavior sanitizer only" OFF)
if((ASAN AND TSAN) OR (UBSAN AND TSAN) OR (ASAN AND UBSAN))
  message(FATAL_ERROR "Use separate sanitizer builds")
endif()
if(ASAN)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
elseif(UBSAN)
  add_compile_options(-fsanitize=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=undefined)
elseif(TSAN)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
include(CTest)
include(cmake/Dependencies.cmake)
add_library(bidder src/bidder.cpp)
target_include_directories(bidder PUBLIC include)
target_include_directories(bidder SYSTEM PUBLIC ${httplib_SOURCE_DIR} ${json_SOURCE_DIR})
target_compile_options(bidder PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
target_link_libraries(bidder PUBLIC Threads::Threads)
add_executable(bidforge src/main.cpp)
target_link_libraries(bidforge PRIVATE bidder)
target_compile_options(bidforge PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(match_bench tests/match_bench.cpp)
target_link_libraries(match_bench PRIVATE bidder)
if(BUILD_TESTING)
  add_executable(bidder_tests tests/bidder_test.cpp)
  target_link_libraries(bidder_tests PRIVATE bidder GTest::gtest_main)
  add_test(NAME bidder_tests COMMAND bidder_tests)
endif()
